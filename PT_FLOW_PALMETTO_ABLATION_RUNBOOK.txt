# 4. Check status:

#    ABL_TAG=cifar10_ptflow_screen \
#    bash PT_FLOW_PALMETTO_ABLATION_RUNBOOK.txt status

#    /scratch/$USER/ptflow_ablation/runs/$ABL_TAG/report/REPORT.md

# ----------------
# This is a screening suite for choosing promising settings. A 5k-step run and

"""Synthetic proposal-family x epsilon x dimension diagnostic.

The analytic potential is smooth and strongly convex. M3 is the exact spectral
norm of its diagonal third-derivative tensor at the prox point. The experiment
reports normalized ESS/K and log-weight spread for naive, recentered-isotropic,
and diagonal-Laplace proposals. This tests the scaling mechanism; it is not an
image-quality experiment.
"""
from __future__ import annotations
import argparse, csv, json, math
from pathlib import Path
import torch

p = argparse.ArgumentParser()
p.add_argument("--out", type=Path, required=True)
p.add_argument("--k", type=int, default=512)
p.add_argument("--repeats", type=int, default=4)
a = p.parse_args()
a.out.mkdir(parents=True, exist_ok=True)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
dims = (32, 128, 512, 2048, 3072, 4096)
eps_values = (0.2, 0.1, 0.05, 0.02, 0.01, 0.005)
families = ("naive", "recentered", "diag_laplace")
kappa, beta, x_value = 0.25, 0.15, 2.0

def prox_scalar():
    y = torch.tensor(x_value, dtype=torch.float64)
    for _ in range(30):
        f = (1 + kappa) * y + beta * torch.sin(y) - x_value
        y = y - f / ((1 + kappa) + beta * torch.cos(y))
    return float(y)

m_scalar = prox_scalar()
h_scalar = (1 + kappa) + beta * math.cos(m_scalar)
m3_exact = abs(beta * math.sin(m_scalar))
rows = []
for d in dims:
    # Float64 is deliberate: the proposal log-density and target exponent can
    # each be large at d=4096 and small epsilon even when their difference is O(1).
    x = torch.full((d,), x_value, device=device, dtype=torch.float64)
    m = torch.full((d,), m_scalar, device=device, dtype=torch.float64)
    for eps in eps_values:
        for family in families:
            ess_values, spread_values = [], []
            for repeat in range(a.repeats):
                gen = torch.Generator(device=device).manual_seed(1000 + d + repeat)
                half = (a.k + 1) // 2
                z0 = torch.randn((half, d), generator=gen, device=device, dtype=torch.float64)
                z = torch.cat((z0, -z0), dim=0)[:a.k]
                if family == "naive":
                    center, scale = x, math.sqrt(2 * eps)
                elif family == "recentered":
                    center, scale = m, math.sqrt(2 * eps)
                else:
                    center, scale = m, math.sqrt(2 * eps / h_scalar)
                y = center.unsqueeze(0) + scale * z
                phi = 0.5 * kappa * y.square().sum(1) + beta * (1 - torch.cos(y)).sum(1)
                f = phi + 0.5 * (x.unsqueeze(0) - y).square().sum(1)
                log_target = -f.double() / (2 * eps)
                log_q = -0.5 * ((y - center.unsqueeze(0)) / scale).double().square().sum(1)
                log_q = log_q - d * math.log(scale)
                lw = log_target - log_q
                l1 = torch.logsumexp(lw, 0)
                l2 = torch.logsumexp(2 * lw, 0)
                ess = torch.exp(2 * l1 - l2 - math.log(a.k)).clamp(0, 1)
                ess_values.append(float(ess))
                spread_values.append(float(lw.std(correction=0)))
            rows.append({
                "family": family, "eps": eps, "d": d, "K": a.k,
                "ess_over_K_mean": sum(ess_values) / len(ess_values),
                "ess_over_K_std": float(torch.tensor(ess_values).std(correction=0)),
                "logw_spread_mean": sum(spread_values) / len(spread_values),
                "M3_exact": m3_exact,
                "d_M3_sq_eps": d * m3_exact * m3_exact * eps,
            })

fields = list(rows[0])
with (a.out / "proposal_eps_d.csv").open("w", newline="", encoding="utf-8") as f:
    w = csv.DictWriter(f, fieldnames=fields); w.writeheader(); w.writerows(rows)
(a.out / "proposal_eps_d.json").write_text(json.dumps(rows, indent=2) + "\n", encoding="utf-8")

try:
    import matplotlib; matplotlib.use("Agg")
    import matplotlib.pyplot as plt
    fig, axes = plt.subplots(2, 3, figsize=(14, 8), sharex=True, sharey=True)
    for ax, d in zip(axes.ravel(), dims):
        for family in families:
            rr = [r for r in rows if r["d"] == d and r["family"] == family]
            ax.plot([r["eps"] for r in rr], [r["ess_over_K_mean"] for r in rr], "o-", label=family)
        ax.set_xscale("log"); ax.invert_xaxis(); ax.set_title(f"d={d}")
        ax.grid(alpha=.3); ax.set_ylim(0, 1.05)
    axes[0,0].legend(fontsize=8); fig.supxlabel("epsilon"); fig.supylabel("normalized ESS/K")
    fig.tight_layout(); fig.savefig(a.out / "proposal_ess_vs_eps_and_d.png", dpi=180); plt.close(fig)
except Exception as exc:
    (a.out / "plot_error.txt").write_text(str(exc), encoding="utf-8")
print(f"wrote {len(rows)} rows to {a.out}")

"""Directional finite-difference M3 proxy on trained epsilon variants."""
from __future__ import annotations
import argparse, csv, gc, json, math, sys
from pathlib import Path
import torch, yaml

p = argparse.ArgumentParser()
p.add_argument("--code", type=Path, required=True)
p.add_argument("--run-root", type=Path, required=True)
p.add_argument("--steps", type=int, required=True)
p.add_argument("--probes", type=int, default=3)
p.add_argument("--fd-step", type=float, default=1e-2)
a = p.parse_args()
sys.path.insert(0, str(a.code.resolve()))
from inference import _load_model
from torch.nn.attention import sdpa_kernel, SDPBackend

root = a.run_root.resolve()
runs = [("base", root / "configs/base.yaml", root / "train/base")]
for eps in (0.2, 0.1, 0.05, 0.02, 0.01):
    name = f"eps_{eps:g}"
    runs.append((name, root / f"configs/{name}.yaml", root / f"train/{name}"))

rows = []
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
for run_id, config, workdir in runs:
    ckpt = workdir / "checkpoints" / f"state_{a.steps:08d}.pt"
    row = {"run_id": run_id, "config": str(config), "ckpt": str(ckpt)}
    try:
        if not ckpt.is_file(): raise FileNotFoundError(ckpt)
        cfg = yaml.safe_load(config.read_text(encoding="utf-8"))
        row["requested_eps_min"] = float(cfg["pt"]["schedule"]["eps_min"])
        model, _, _, _, potential, _, trained_eps = _load_model(
            str(ckpt), str(config), want_potential=True)
        row["checkpoint_eps"] = float(trained_eps)
        gen = torch.Generator(device=device).manual_seed(1234)
        labels = torch.zeros(1, dtype=torch.long, device=device)
        with torch.no_grad():
            out = model(c=labels, cfg_scale=1.0, deterministic=True, train=False, rng=gen)
            m = out["samples"].detach().float()
        values = []
        for probe in range(a.probes):
            g = torch.Generator(device=device).manual_seed(9000 + probe)
            v = torch.empty_like(m).bernoulli_(0.5, generator=g).mul_(2).sub_(1)
            v = v / v.flatten(1).norm(dim=1).view(-1, *([1] * (v.ndim - 1)))
            curvatures = []
            for sign in (-1.0, 1.0):
                x = (m + sign * a.fd_step * v).detach().requires_grad_(True)
                with torch.enable_grad(), sdpa_kernel(SDPBackend.MATH):
                    phi = potential.phi(x, labels).sum()
                    grad = torch.autograd.grad(phi, x, create_graph=True)[0]
                    hv = torch.autograd.grad((grad * v).sum(), x)[0]
                    curvatures.append(float((hv * v).sum().detach()))
            values.append(abs(curvatures[1] - curvatures[0]) / (2 * a.fd_step))
        row.update(
            directional_M3_proxy_mean=sum(values) / len(values),
            directional_M3_proxy_max=max(values), probes=a.probes,
            fd_step=a.fd_step, status="ok")
        del model, potential, out, m
        gc.collect()
        if torch.cuda.is_available(): torch.cuda.empty_cache()
    except Exception as exc:
        row.update(status="error", error=f"{type(exc).__name__}: {exc}")
    rows.append(row)

outdir = root / "results"; outdir.mkdir(parents=True, exist_ok=True)
(outdir / "m3_vs_eps.json").write_text(json.dumps(rows, indent=2) + "\n", encoding="utf-8")
fields = sorted({k for r in rows for k in r})
